function mapDbSegment(row) {
  return {
    id: row.id,
    name: row.name,
    status: row.status,
    length: row.length,
    crossArea: row.cross_area,
    dimensions: row.dimensions,
    officer: row.officer,
    lastCheck: row.last_check,
    notes: row.notes,
    coords: row.coords || [],
    photos: row.photos || [],
    history: row.history || []
  };
}

function mapDbReport(row) {
  return {
    id: row.id,
    location: row.location,
    desc: row.description,
    dateSubmitted: row.date_submitted,
    status: row.status,
    verified: Boolean(row.verified),
    verifiedAt: row.verified_at,
    verifiedBy: row.verified_by,
    hasUpdate: Boolean(row.has_update),
    isNewForInspector: Boolean(row.is_new_for_inspector),
    coords: row.coords || [10.5247, 124.0285],
    photos: row.photos || []
  };
}

async function fetchCloudState() {
  try {
    const [segRes, repRes, syncRes] = await Promise.all([
      sbClient.from('segments').select('*').order('id', { ascending: true }),
      sbClient.from('citizen_reports').select('*').order('created_at', { ascending: false }),
      sbClient.from('sync_meta').select('*').eq('key', 'suba_network').maybeSingle()
    ]);

    if (segRes.error) throw segRes.error;
    if (repRes.error) throw repRes.error;

    // Seed initial segments once if table is empty
    let segments = (segRes.data || []).map(mapDbSegment);
    if (segments.length === 0) {
      await sbClient.from('segments').upsert(SEED_SEGMENTS);
      segments = SEED_SEGMENTS.map(mapDbSegment);
    }

    const allReports = (repRes.data || []).map(mapDbReport);
    const citizenReports = allReports.filter(r => !r.verified);
    const processedReportHistory = allReports.filter(r => r.verified);
    const lastSync = syncRes.data?.last_field_sync || 'Oct 27, 2026 - 11:54 PM';

    const cloudData = { segments, citizenReports, processedReportHistory, lastSync };
    localStorage.setItem(LOCAL_CACHE_KEY, JSON.stringify(cloudData));
    return cloudData;
  } catch (err) {
    console.warn('Using cached state due to network/Supabase error:', err.message);
    const cached = localStorage.getItem(LOCAL_CACHE_KEY);
    return cached ? JSON.parse(cached) : {
      segments: SEED_SEGMENTS.map(mapDbSegment),
      citizenReports: [],
      processedReportHistory: [],
      lastSync: 'Oct 27, 2026 - 11:54 PM'
    };
  }
}

async function upsertSegmentCloud(seg, syncTimestamp) {
  const payload = {
    id: seg.id,
    name: seg.name,
    status: seg.status,
    length: seg.length,
    cross_area: seg.crossArea,
    dimensions: seg.dimensions,
    officer: seg.officer,
    last_check: seg.lastCheck || syncTimestamp,
    notes: seg.notes || '',
    coords: seg.coords,
    photos: seg.photos || [],
    history: seg.history || [],
    updated_at: new Date().toISOString()
  };

  const { error } = await sbClient.from('segments').upsert(payload);
  if (error) throw error;

  await sbClient.from('sync_meta').upsert({
    key: 'suba_network',
    last_field_sync: syncTimestamp,
    updated_at: new Date().toISOString()
  });
}

async function deleteSegmentCloud(segId, syncTimestamp) {
  const { error } = await sbClient.from('segments').delete().eq('id', segId);
  if (error) throw error;

  await sbClient.from('sync_meta').upsert({
    key: 'suba_network',
    last_field_sync: syncTimestamp,
    updated_at: new Date().toISOString()
  });
}

async function insertCitizenReportCloud(rep) {
  const { error } = await sbClient.from('citizen_reports').insert({
    id: rep.id,
    location: rep.location,
    description: rep.desc,
    date_submitted: rep.dateSubmitted,
    status: rep.status,
    verified: rep.verified,
    verified_at: rep.verifiedAt,
    verified_by: rep.verifiedBy,
    has_update: rep.hasUpdate,
    is_new_for_inspector: rep.isNewForInspector,
    coords: rep.coords,
    photos: rep.photos || []
  });
  if (error) throw error;
}

async function updateCitizenReportCloud(rep) {
  const { error } = await sbClient.from('citizen_reports').update({
    location: rep.location,
    description: rep.desc,
    status: rep.status,
    verified: rep.verified,
    verified_at: rep.verifiedAt,
    verified_by: rep.verifiedBy,
    has_update: rep.hasUpdate,
    is_new_for_inspector: rep.isNewForInspector,
    photos: rep.photos || []
  }).eq('id', rep.id);
  if (error) throw error;
}

function subscribeToCloudChanges(onChange) {
  return sbClient
    .channel('drainwatch-live-changes')
    .on('postgres_changes', { event: '*', schema: 'public', table: 'segments' }, onChange)
    .on('postgres_changes', { event: '*', schema: 'public', table: 'citizen_reports' }, onChange)
    .on('postgres_changes', { event: '*', schema: 'public', table: 'sync_meta' }, onChange)
    .subscribe();
}
